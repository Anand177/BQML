    WARM_START = { TRUE | FALSE },

    DATA_SPLIT_COL = string_value,

    NUM_TRIALS = int64_value,                     -- hyperparameter tuning
    MAX_PARALLEL_TRIALS = int64_value,            -- hyperparameter tuning
    HPARAM_TUNING_OBJECTIVES = string_array,      -- hyperparameter tuning
    HPARAM_TUNING_ALGORITHM = { 'VIZIER_DEFAULT' | 'RANDOM' | 'GRID' }
)]
AS query_statement
